// Disciplina   : Introdução a lógica de programação
// Professor   : Larissa Ribeiro
// Descrição   : Encontra o menor número entre 10 inseridos
// Autor(a)    : Arthur Santiago Mayolino e Eduardo Sberse
// Data atual  : 07/10/2024
Var
// Seção de Declarações das variáveis 
   number, smallest: real
   counter: inteiro

Inicio
// Seção de Comandos, procedimento, funções, operadores, etc... 
   escreval("Insira um número")
   leia(smallest)
   counter <- 1
   repita
         escreval("Insira outro número")
         leia(number)
         se number < smallest entao
            smallest <- number
         fimse
         counter <- counter + 1
   ate counter = 10
   escreval("O menor é ", smallest)
